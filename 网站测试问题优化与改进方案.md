# 1. 统一错误处理装饰器
def api_error_handler(func):
    def wrapper(*args, **kwargs):
        try:
            return func(*args, **kwargs)
        except ValidationError as e:
            return JsonResponse({
                'error': 'VALIDATION_ERROR',
                'message': str(e),
                'code': 400
            }, status=400)
        except FileNotFoundError as e:
            return JsonResponse({
                'error': 'FILE_NOT_FOUND',
                'message': '文件不存在',
                'code': 404
            }, status=404)
        except Exception as e:
            logger.error(f"API Error: {str(e)}")
            return JsonResponse({
                'error': 'INTERNAL_ERROR',
                'message': '服务器内部错误',
                'code': 500
            }, status=500)
    return wrapper

# 2. PDF转换API优化
@api_error_handler
def pdf_converter_api(request):
    if request.method != 'POST':
        return JsonResponse({'error': 'METHOD_NOT_ALLOWED', 'code': 405}, status=405)
    
    file = request.FILES.get('file')
    if not file:
        return JsonResponse({
            'error': 'MISSING_FILE',
            'message': '请上传PDF文件',
            'code': 400
        }, status=400)
    
    # 文件类型验证
    if not file.name.lower().endswith('.pdf'):
        return JsonResponse({
            'error': 'INVALID_FILE_TYPE',
            'message': '只支持PDF文件',
            'code': 400
        }, status=400)
    
    # 文件大小验证
    if file.size > 10 * 1024 * 1024:  # 10MB
        return JsonResponse({
            'error': 'FILE_TOO_LARGE',
            'message': '文件大小不能超过10MB',
            'code': 400
        }, status=400)
    
    try:
        result = convert_pdf_to_format(file, request.POST.get('format', 'docx'))
        return JsonResponse({
            'success': True,
            'download_url': result['download_url'],
            'file_name': result['file_name']
        })
    except ConversionError as e:
        return JsonResponse({
            'error': 'CONVERSION_FAILED',
            'message': f'转换失败: {str(e)}',
            'code': 422
        }, status=422)

def cache_result(timeout=300):
    def decorator(func):
        def wrapper(*args, **kwargs):
            cache_key = f"{func.__name__}:{hash(str(args) + str(kwargs))}"
            result = cache.get(cache_key)
            
            if result is None:
                result = func(*args, **kwargs)
                cache.set(cache_key, result, timeout)
            
            return result
        return wrapper
    return decorator

# 4. BMI计算API优化
@cache_result(timeout=600)  # 缓存10分钟
def calculate_bmi_optimized(height, weight):
    """优化的BMI计算，带缓存"""
    # 使用数据库索引优化查询
    bmi_result = BMICalculation.objects.filter(
        height=height, 
        weight=weight
    ).first()
    
    if bmi_result:
        return bmi_result.result
    
    # 计算并保存结果
    bmi = weight / ((height / 100) ** 2)
    BMICalculation.objects.create(
        height=height,
        weight=weight,
        result=bmi
    )
    
    return bmi

class BasePage:
    def __init__(self, driver):
        self.driver = driver

        return self.wait.until(EC.presence_of_element_located(locator))

        element = self.find_element(locator)
        self.wait.until(EC.element_to_be_clickable(locator))

    def input_text(self, locator, text):

# 1. 输入验证装饰器
def validate_input(allowed_fields=None, max_length=None):
    def decorator(func):
        def wrapper(request, *args, **kwargs):
            # 检查请求方法
            if request.method not in ['POST', 'PUT', 'PATCH']:
                return func(request, *args, **kwargs)
            
            # 获取输入数据
            if request.content_type == 'application/json':
                data = request.json
            else:
                data = request.POST
            
            # 字段白名单验证
            if allowed_fields:
                for field in data.keys():
                    if field not in allowed_fields:
                        return JsonResponse({
                            'error': 'INVALID_FIELD',
                            'message': f'不允许的字段: {field}',
                            'code': 400
                        }, status=400)
            
            # 长度验证
            if max_length:
                for field, value in data.items():
                    if isinstance(value, str) and len(value) > max_length:
                        return JsonResponse({
                            'error': 'FIELD_TOO_LONG',
                            'message': f'字段 {field} 长度超过限制',
                            'code': 400
                        }, status=400)
            
            return func(request, *args, **kwargs)
        return wrapper
    return decorator

class SQLInjectionProtection:
    @staticmethod
    def sanitize_input(input_data):
        """清理输入数据"""
        if isinstance(input_data, str):
            # 移除危险字符
            dangerous_chars = ["'", '"', ';', '--', '/*', '*/', 'xp_', 'sp_']
            for char in dangerous_chars:
                input_data = input_data.replace(char, '')
            
            # 长度限制
            if len(input_data) > 1000:
                input_data = input_data[:1000]
        
        return input_data
    
    @staticmethod
    def validate_sql_input(input_data):
        """验证SQL输入"""
        sql_patterns = [
            r'union\s+select',
            r'drop\s+table',
            r'delete\s+from',
            r'insert\s+into',
            r'update\s+set',
            r'exec\s*\(',
            r'execute\s*\(',
        ]
        
        for pattern in sql_patterns:
            if re.search(pattern, input_data, re.IGNORECASE):
                raise ValidationError("检测到潜在的SQL注入攻击")

    def sanitize_html(html_content):

class DefectManager:
    def __init__(self):
        self.defects = []
        self.severity_levels = {
            'CRITICAL': 1,
            'HIGH': 2,
            'MEDIUM': 3,
            'LOW': 4
        }
    
    def create_defect(self, title, description, severity, test_case_id):
        """创建缺陷记录"""
        defect = {
            'id': f"DEF-{len(self.defects) + 1:04d}",
            'title': title,
            'description': description,
            'severity': severity,
            'status': 'OPEN',
            'test_case_id': test_case_id,
            'created_at': datetime.now(),
            'assigned_to': None,
            'resolution': None
        }
        
        self.defects.append(defect)
        return defect
    
    def get_defects_by_severity(self, severity):
        """按严重程度获取缺陷"""
        return [d for d in self.defects if d['severity'] == severity]
    
    def generate_defect_report(self):
        """生成缺陷报告"""
        report = {
            'total_defects': len(self.defects),
            'open_defects': len([d for d in self.defects if d['status'] == 'OPEN']),
            'closed_defects': len([d for d in self.defects if d['status'] == 'CLOSED']),
            'by_severity': {}
        }
        
        for severity in self.severity_levels.keys():
            report['by_severity'][severity] = len(
                [d for d in self.defects if d['severity'] == severity]
            )
        
        return report
