#!/usr/bin/env python3
"""
接口测试脚本 - shenyiqing.xin
测试REST API端点的功能和性能
"""

import requests
import json
import time
import pytest
from urllib.parse import urljoin

class APITester:
    def __init__(self, base_url="https://shenyiqing.xin"):
        self.base_url = base_url
        self.session = requests.Session()
        self.session.headers.update({
            'Content-Type': 'application/json',
            'User-Agent': 'API-Tester/1.0'
        })
        self.api_results = []
    
    def test_api_endpoint(self, endpoint, method='GET', data=None, expected_status=200):
        """测试单个API端点"""
        url = urljoin(self.base_url, endpoint)
        start_time = time.time()
        
        try:
            if method.upper() == 'GET':
                response = self.session.get(url, timeout=10)
            elif method.upper() == 'POST':
                response = self.session.post(url, json=data, timeout=10)
            elif method.upper() == 'PUT':
                response = self.session.put(url, json=data, timeout=10)
            elif method.upper() == 'DELETE':
                response = self.session.delete(url, timeout=10)
            
            end_time = time.time()
            response_time = end_time - start_time
            
            result = {
                'endpoint': endpoint,
                'method': method,
                'status_code': response.status_code,
                'response_time': response_time,
                'expected_status': expected_status,
                'success': response.status_code == expected_status,
                'content_type': response.headers.get('content-type', ''),
                'content_length': len(response.content)
            }
            
            # 尝试解析JSON响应
            try:
                result['json_data'] = response.json()
            except:
                result['json_data'] = None
            
            self.api_results.append(result)
            
            if result['success']:
                print(f"✅ {method} {endpoint} - {response.status_code} ({response_time:.3f}s)")
            else:
                print(f"❌ {method} {endpoint} - {response.status_code} (期望: {expected_status})")
            
            return result
            
        except requests.exceptions.Timeout:
            print(f"⏰ {method} {endpoint} - 请求超时")
            self.api_results.append({
                'endpoint': endpoint,
                'method': method,
                'status_code': 'TIMEOUT',
                'response_time': 10.0,
                'success': False,
                'error': 'Request timeout'
            })
        except Exception as e:
            print(f"💥 {method} {endpoint} - 异常: {str(e)}")
            self.api_results.append({
                'endpoint': endpoint,
                'method': method,
                'status_code': 'ERROR',
                'response_time': 0,
                'success': False,
                'error': str(e)
            })

        # 测试用户资料API

        import threading
        import requests
        import time
        
        print("🔄 测试并发请求...")
        
        def make_request():
            try:
                start_time = time.time()
                response = requests.get(f"{self.base_url}/", timeout=10)
                end_time = time.time()
                return {
                    'status_code': response.status_code,
                    'response_time': end_time - start_time,
                    'success': response.status_code == 200
                }
            except:
                return {
                    'status_code': 'ERROR',
                    'response_time': 10.0,
                    'success': False
                }
        
        # 测试不同并发级别
        concurrent_levels = [1, 5, 10, 20, 50]
        
        for level in concurrent_levels:
            print(f"🚀 测试 {level} 个并发请求...")
            
            threads = []
            results = []
            
            start_time = time.time()
            
            for _ in range(level):
                thread = threading.Thread(target=lambda: results.append(make_request()))
                threads.append(thread)
                thread.start()
            
            for thread in threads:
                thread.join()
            
            end_time = time.time()
            total_time = end_time - start_time
            
            successful = len([r for r in results if r['success']])
            avg_response_time = sum([r['response_time'] for r in results]) / len(results)
            
            concurrent_result = {
                'concurrent_level': level,
                'total_time': total_time,
                'successful_requests': successful,
                'success_rate': successful / level * 100,
                'avg_response_time': avg_response_time,
                'requests_per_second': level / total_time
            }
            
            self.results.append(concurrent_result)
            
            print(f"📈 并发 {level}: 成功率 {successful/level*100:.1f}%, "

"""
安全测试脚本 - shenyiqing.xin
基于OWASP Top 10进行安全漏洞检测

import requests
import json
import time
import re
from urllib.parse import urljoin, quote

import socket

class SecurityTester:
    def __init__(self, base_url="https://shenyiqing.xin"):
        self.base_url = base_url
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Security-Tester/1.0'
        })
        self.security_results = []
    
    def test_sql_injection(self):
        """测试SQL注入漏洞"""
        print("🔍 测试SQL注入漏洞...")
        
        # SQL注入测试载荷
        sql_payloads = [
            "' OR '1'='1",
            "' UNION SELECT * FROM users--",
            "'; DROP TABLE users; --",
            "' OR 1=1--",
            "admin'--",
            "' OR 'x'='x"
        ]
        
        # 测试端点
        test_endpoints = [
            '/api/users/login/',
            '/api/fitness/bmi/',
            '/api/generate-testcases/',
            '/job-search/api/start/'
        ]
        
        for endpoint in test_endpoints:
            for payload in sql_payloads:
                try:
                    # 测试POST请求
                    test_data = {
                        'username': payload,
                        'password': 'test',
                        'height': payload,
                        'weight': payload
                    }
                    
                    response = self.session.post(
                        urljoin(self.base_url, endpoint),
                        json=test_data,
                        timeout=5
                    )
                    
                    # 检查响应中的SQL错误信息
                    sql_errors = [
                        'mysql_fetch',
                        'ORA-01756',
                        'Microsoft OLE DB',
                        'ODBC SQL Server Driver',
                        'PostgreSQL query failed',
                        'Warning: mysql_',
                        'MySQLSyntaxErrorException',
                        'valid MySQL result',
                        'MySqlClient',
                        'SQLException'
                    ]
                    
                    response_text = response.text.lower()
                    for error in sql_errors:
                        if error.lower() in response_text:
                            print(f"🚨 SQL注入漏洞发现: {endpoint}")
                            self.security_results.append({
                                'vulnerability': 'SQL Injection',
                                'endpoint': endpoint,
                                'payload': payload,
                                'severity': 'HIGH',
                                'description': f'SQL错误信息泄露: {error}'
                            })
                            break
                    
                    # 检查异常响应时间（可能的盲注）
                    if response.elapsed.total_seconds() > 5:
                        print(f"⚠️ 可能的盲SQL注入: {endpoint}")
                        self.security_results.append({
                            'vulnerability': 'Blind SQL Injection',
                            'endpoint': endpoint,
                            'payload': payload,
                            'severity': 'MEDIUM',
                            'description': '响应时间异常，可能存在盲注'
                        })
                
                except Exception as e:
                    pass
        
        print(f"✅ SQL注入测试完成，发现 {len([r for r in self.security_results if 'SQL' in r['vulnerability']])} 个问题")
    
    def test_xss_vulnerability(self):
        """测试跨站脚本攻击(XSS)漏洞"""
        print("🔍 测试XSS漏洞...")
        
        # XSS测试载荷
        xss_payloads = [
            "<script>alert('XSS')</script>",
            "javascript:alert('XSS')",
            "<img src=x onerror=alert('XSS')>",
            "<svg onload=alert('XSS')>",
            "';alert('XSS');//",
            "\"><script>alert('XSS')</script>",
            "<iframe src=javascript:alert('XSS')>",
            "<body onload=alert('XSS')>"
        ]
        
        # 测试参数
        test_params = ['username', 'email', 'comment', 'search', 'message']
        
        for param in test_params:
            for payload in xss_payloads:
                try:
                    # 测试GET请求
                    test_url = f"{self.base_url}/?{param}={quote(payload)}"
                    response = self.session.get(test_url, timeout=5)
                    
                    # 检查响应中是否包含未转义的载荷
                    if payload in response.text and '<script>' in payload:
                        print(f"🚨 XSS漏洞发现: GET参数 {param}")
                        self.security_results.append({
                            'vulnerability': 'Cross-Site Scripting (XSS)',
                            'endpoint': f'GET ?{param}',
                            'payload': payload,
                            'severity': 'HIGH',
                            'description': '用户输入未正确转义'
                        })
                    
                    # 测试POST请求
                    test_data = {param: payload}
                    response = self.session.post(
                        f"{self.base_url}/api/users/login/",
                        json=test_data,
                        timeout=5
                    )
                    
                    if payload in response.text and '<script>' in payload:
                        print(f"🚨 XSS漏洞发现: POST参数 {param}")
                        self.security_results.append({
                            'vulnerability': 'Cross-Site Scripting (XSS)',
                            'endpoint': f'POST {param}',
                            'payload': payload,
                            'severity': 'HIGH',
                            'description': '用户输入未正确转义'
                        })
                
                except Exception as e:
                    pass
        
        print(f"✅ XSS测试完成，发现 {len([r for r in self.security_results if 'XSS' in r['vulnerability']])} 个问题")
    
    def test_csrf_vulnerability(self):
        """测试跨站请求伪造(CSRF)漏洞"""
        print("🔍 测试CSRF漏洞...")
        
        # 测试CSRF保护的端点
        csrf_endpoints = [

                # 不带CSRF token的请求

                    json={'test': 'data'},
                    timeout=5
                )
                
                # 检查是否返回CSRF错误
                if response.status_code == 403 and 'csrf' in response.text.lower():
                    print(f"✅ CSRF保护正常: {endpoint}")
                elif response.status_code == 200:
                    print(f"🚨 CSRF漏洞发现: {endpoint}")
                    self.security_results.append({
                        'vulnerability': 'Cross-Site Request Forgery (CSRF)',
                        'endpoint': endpoint,
                        'severity': 'MEDIUM',
                        'description': '缺少CSRF保护'
                    })
            
            except Exception as e:
                pass
        
        print(f"✅ CSRF测试完成，发现 {len([r for r in self.security_results if 'CSRF' in r['vulnerability']])} 个问题")
    
    def test_sensitive_data_exposure(self):
        """测试敏感信息泄露"""
        print("🔍 测试敏感信息泄露...")
        
        # 检查敏感文件
        sensitive_files = [
            '/.env',
            '/config.py',
            '/settings.py',
            '/database.yml',
            '/.git/config',
            '/robots.txt',
            '/sitemap.xml',
            '/admin/',
            '/phpmyadmin/',
            '/.htaccess'

        for file_path in sensitive_files:
            try:
                response = self.session.get(
                    urljoin(self.base_url, file_path),
                    timeout=5
                )
                
                if response.status_code == 200:
                    content = response.text.lower()
                    
                    # 检查敏感信息
                    sensitive_patterns = [
                        'password',
                        'secret',
                        'key',
                        'token',
                        'api_key',
                        'database',
                        'mysql',
                        'postgresql'
                    ]
                    
                    for pattern in sensitive_patterns:
                        if pattern in content:
                            print(f"🚨 敏感信息泄露: {file_path}")
                            self.security_results.append({
                                'vulnerability': 'Sensitive Data Exposure',
                                'endpoint': file_path,
                                'severity': 'HIGH',
                                'description': f'发现敏感信息: {pattern}'
                            })
                            break
            
            except Exception as e:
                pass
        
        print(f"✅ 敏感信息泄露测试完成，发现 {len([r for r in self.security_results if 'Sensitive' in r['vulnerability']])} 个问题")

    def test_security_headers(self):
        """测试安全头"""
        print("🔍 测试安全头...")
        
        try:
            response = self.session.get(self.base_url, timeout=5)
            headers = response.headers
            
            # 检查重要的安全头
            security_headers = {
                'X-Content-Type-Options': 'nosniff',
                'X-Frame-Options': ['DENY', 'SAMEORIGIN'],
                'X-XSS-Protection': '1; mode=block',
                'Strict-Transport-Security': 'max-age=',
                'Content-Security-Policy': None,
                'Referrer-Policy': None
            }
            
            missing_headers = []
            for header, expected_value in security_headers.items():
                if header not in headers:
                    missing_headers.append(header)
                    print(f"⚠️ 缺少安全头: {header}")
                elif expected_value and expected_value not in str(headers[header]):
                    print(f"⚠️ 安全头配置不当: {header}")
            
            if missing_headers:
                self.security_results.append({
                    'vulnerability': 'Missing Security Headers',
                    'severity': 'MEDIUM',
                    'description': f'缺少安全头: {", ".join(missing_headers)}'
                })
            
            print(f"✅ 安全头测试完成，发现 {len(missing_headers)} 个问题")
        
        except Exception as e:
            print(f"❌ 安全头测试失败: {str(e)}")
    
    def test_ssl_tls_security(self):
        """测试SSL/TLS安全"""
        print("🔍 测试SSL/TLS安全...")
        
        try:
            # 使用nmap检查SSL配置
            result = subprocess.run([
                'nmap', '--script', 'ssl-enum-ciphers', '-p', '443', 'shenyiqing.xin'
            ], capture_output=True, text=True, timeout=30)
            
            if result.returncode == 0:
                output = result.stdout
                
                # 检查弱加密套件
                weak_ciphers = ['RC4', 'DES', 'MD5', 'SHA1']
                for cipher in weak_ciphers:
                    if cipher in output:
                        print(f"🚨 发现弱加密套件: {cipher}")
                        self.security_results.append({
                            'vulnerability': 'Weak SSL/TLS Configuration',
                            'severity': 'HIGH',
                            'description': f'使用弱加密套件: {cipher}'
                        })
                
                print("✅ SSL/TLS安全测试完成")
            else:
                print("⚠️ SSL/TLS测试需要安装nmap")
        
        except Exception as e:
            print(f"⚠️ SSL/TLS测试跳过: {str(e)}")

        medium_severity = len([r for r in self.security_results if r.get('severity') == 'MEDIUM'])

        # 安全评分

        total_score -= low_severity * 5
        total_score = max(0, total_score)

        return self.security_results
